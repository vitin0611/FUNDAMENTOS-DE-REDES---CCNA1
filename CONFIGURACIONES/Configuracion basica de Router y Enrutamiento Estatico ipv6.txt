enable
configure terminal
hostname HIGUEY
ipv6 unicast-routing

# Pool DHCPv6 para Estudiantes
ipv6 dhcp pool POOL-ESTUDIANTES
 address prefix 2003:BABA:506:1::/64
 dns-server 2003:BABA:506:4::5
 exit

# Interfaz LAN - RED DE ESTUDIANTES
interface GigabitEthernet0/0
 ipv6 address 2003:BABA:506:1::1/64
 ipv6 address FE80::1:1 link-local
 ipv6 dhcp server POOL-ESTUDIANTES
 ipv6 nd managed-config-flag
 no shutdown
 exit

# Interfaz WAN hacia PUNTA_CANTA
interface Serial0/0/0
 ipv6 address 2003:BABA:506:A::1/64
 ipv6 address FE80::A:1 link-local
 no shutdown
 exit

# Enrutamiento Estático (Hacia todas las redes remotas)
ipv6 route 2003:BABA:506:2::/64 2003:BABA:506:A::2
ipv6 route 2003:BABA:506:3::/64 2003:BABA:506:A::2
ipv6 route 2003:BABA:506:4::/64 2003:BABA:506:A::2
ipv6 route 2003:BABA:506:B::/64 2003:BABA:506:A::2
